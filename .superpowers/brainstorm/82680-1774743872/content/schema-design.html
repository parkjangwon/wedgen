템플릿 스키마 (schema.json) 설계

각 템플릿이 선언하는 커스터마이징 옵션 구조

스키마 구조 개요

schema.json 최상위 구조
{
  "id": "elegant-classic",
  "name": "Elegant Classic",
  "description": "클래식한 세로 스크롤 청첩장",
  "version": "1.0.0",
  "thumbnail": "thumbnail.png",
  "interaction": "scroll",  // "scroll" | "page" | "parallax"

  "sections": {
    // 사용 가능한 섹션 목록 + 기본 순서
    "available": ["hero","greeting","gallery","info","guestbook","rsvp"],
    "defaultOrder": ["hero","greeting","gallery","info","guestbook","rsvp"],
    "required": ["hero","info"]  // 필수 (삭제 불가)
  },

  "fields": {
    // 탭별로 그룹화된 입력 필드 정의
    "기본 정보": { ... },
    "갤러리": { ... },
    "예식 정보": { ... },
    "방명록 & RSVP": { ... },
    "스타일": { ... }
  },

  "style": {
    // 템플릿이 허용하는 스타일 커스터마이징 옵션
    "colorSchemes": [...],
    "fonts": [...],
    "animations": [...]
  },

  "i18n": {
    "defaultLocale": "ko",
    "supported": ["ko"]
  }
}

fields 상세: 필드 타입 시스템

지원하는 필드 타입 및 예시
